cmake_minimum_required(VERSION 3.20)
project(lw_Web2Harmony VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The desktop packer currently targets Windows x64")
endif()
include(CTest)
include(FetchContent)
set(SPDLOG_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SPDLOG_INSTALL OFF CACHE BOOL "" FORCE)
set(SPDLOG_WCHAR_FILENAMES ON CACHE BOOL "" FORCE)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.deps/spdlog.tar.gz")
  set(SPDLOG_ARCHIVE "${CMAKE_CURRENT_SOURCE_DIR}/.deps/spdlog.tar.gz")
else()
  set(SPDLOG_ARCHIVE "https://github.com/gabime/spdlog/archive/refs/tags/v1.17.0.tar.gz")
endif()
FetchContent_Declare(spdlog URL "${SPDLOG_ARCHIVE}"
  URL_HASH SHA256=d8862955c6d74e5846b3f580b1605d2428b11d97a410d86e2fb13e857cd3a744
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(spdlog)
add_library(harmony_core packer/core/Json.cpp packer/core/Hash.cpp
  packer/core/Logging.cpp packer/core/ProcessRunner.cpp packer/core/IconGenerator.cpp
  packer/core/HarmonyProject.cpp)
target_include_directories(harmony_core PUBLIC packer)
target_link_libraries(harmony_core PUBLIC spdlog::spdlog bcrypt ole32 windowscodecs)
target_compile_features(harmony_core PUBLIC cxx_std_17)
target_compile_definitions(harmony_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_definitions(harmony_core PUBLIC LW_VERSION="${PROJECT_VERSION}")
add_executable(lw.Web2Harmony packer/main.cpp)
target_link_libraries(lw.Web2Harmony PRIVATE harmony_core)
add_executable(lw.Web2Harmony.GUI WIN32 packer/gui/GuiMain.cpp)
target_link_libraries(lw.Web2Harmony.GUI PRIVATE harmony_core comctl32 shell32 comdlg32)
set(LW_APP_ICON_RESOURCE "${CMAKE_CURRENT_SOURCE_DIR}/assets/app.ico")
foreach(target lw.Web2Harmony lw.Web2Harmony.GUI)
  set(LW_VERSION_FILE_DESCRIPTION "lw.Web2Harmony Web to HarmonyOS")
  set(LW_VERSION_INTERNAL_NAME "${target}")
  set(LW_VERSION_ORIGINAL_FILENAME "${target}.exe")
  configure_file(packer/windows/version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/${target}.rc" @ONLY)
  target_sources(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${target}.rc")
endforeach()
if(BUILD_TESTING)
  add_executable(harmony_tests tests/ProjectTests.cpp)
  target_link_libraries(harmony_tests PRIVATE harmony_core)
  add_test(NAME project_generator COMMAND harmony_tests "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME gui_smoke COMMAND lw.Web2Harmony.GUI --smoke-test)
  find_program(LW_WINDOWS_POWERSHELL NAMES powershell.exe
    PATHS "$ENV{SystemRoot}/System32/WindowsPowerShell/v1.0" NO_DEFAULT_PATH REQUIRED)
  add_test(NAME signing_credentials COMMAND "${LW_WINDOWS_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass
    -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/SigningTests.ps1" -RepositoryRoot "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME signing_module_path_isolation COMMAND "${LW_WINDOWS_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass
    -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/SigningTests.ps1" -RepositoryRoot "${CMAKE_CURRENT_SOURCE_DIR}" -PoisonModulePath)
  set_tests_properties(gui_smoke PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
if(MSVC)
  foreach(target harmony_core lw.Web2Harmony lw.Web2Harmony.GUI)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endforeach()
  if(BUILD_TESTING)
    target_compile_options(harmony_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
endif()
